#Audio Function 1
def hide_audio_message(audio_path, message, output_path):
    """Hide a message in an audio file using LSB."""
    audio = wave.open(audio_path, 'rb')
    params = audio.getparams()
    frames = np.frombuffer(audio.readframes(audio.getnframes()), dtype=np.int16)
    audio.close()

    # Convert message to binary and add a delimiter
    message += "<<<EOF>>>"  # Unique delimiter to identify message end
    binary_message = ''.join(format(ord(char), '08b') for char in message)

    if len(binary_message) > len(frames):
        raise ValueError("Message is too long to encode in the selected audio file.")

    # Embed binary message into audio frames
    for i in range(len(binary_message)):
        frames[i] = (frames[i] & ~1) | int(binary_message[i])

    # Save the modified audio
    encoded_audio = wave.open(output_path, 'wb')
    encoded_audio.setparams(params)
    encoded_audio.writeframes(frames.tobytes())
    encoded_audio.close()

def encrypt_message(message):
    """Encrypt the message using base64 encoding for simple obfuscation."""
    encoded_bytes = base64.b64encode(message.encode('utf-8'))
    return encoded_bytes.decode('utf-8')

def decrypt_message(encrypted_message):
    """Decrypt the message using base64 decoding."""
    decoded_bytes = base64.b64decode(encrypted_message.encode('utf-8'))
    return decoded_bytes.decode('utf-8')

def hide_audio_message(audio_path, message, output_path):
    """
        Hides a message in an audio file by modifying the least significant bit of audio samples.

        :param audio_path: Path to the original audio file.
        :param message: Message to hide in the audio file.
        :param output_path: Path to save the audio file with the hidden message.
        """
    # Convert the message to binary
    message_binary = ''.join(format(ord(char), '08b') for char in message)
    message_length = len(message_binary)

    # Open the audio file
    with wave.open(audio_path, 'rb') as audio:
        params = audio.getparams()  # Get audio file parameters
        num_channels = params.nchannels
        sample_width = params.sampwidth
        num_frames = params.nframes

        # Ensure the file uses 16-bit samples
        if sample_width != 2:
            raise ValueError("Only 16-bit WAV files are supported.")

        # Read all frames and unpack into a list of samples
        frames = audio.readframes(num_frames)
        frame_list = list(struct.unpack("<" + "h" * num_frames * num_channels, frames))

    # Modify LSB of audio samples to embed the message
    if len(message_binary) > len(frame_list):
        raise ValueError("Message is too long to fit in the selected audio file.")

    # Embed message length in the first 32 samples
    length_binary = format(message_length, '032b')
    for i in range(32):
        frame_list[i] = (frame_list[i] & ~1) | int(length_binary[i])

    # Embed the message in subsequent samples
    for i in range(message_length):
        frame_list[32 + i] = (frame_list[32 + i] & ~1) | int(message_binary[i])

    # Pack the modified frames
    modified_frames = struct.pack("<" + "h" * len(frame_list), *frame_list)

    # Save the modified audio
    with wave.open(output_path, 'wb') as modified_audio:
        modified_audio.setparams(params)
        modified_audio.writeframes(modified_frames)

    print(f"Audio file saved as {output_path}.")


def Show():
    global filename
    if filename.lower().endswith((".png", ".jpg")):
        # Reveal hidden message in image
        try:
            clear_message = lsb.reveal(filename)
            if clear_message is None or clear_message.strip() == "":
                text1.delete(1.0, END)
                text1.insert(END, "No hidden message found in the image!")
            else:
                text1.delete(1.0, END)
                text1.insert(END, clear_message)
        except Exception as e:
            text1.delete(1.0, END)
            text1.insert(END, f"Error: {str(e)}")

    elif filename.lower().endswith(".txt"):
        # Read and display decrypted message from text file
        try:
            with open(filename, "r") as file:
                encrypted_message = file.read()
            hidden_message = decrypt_message(encrypted_message)
            text1.delete(1.0, END)
            text1.insert(END, hidden_message)
        except Exception as e:
            text1.delete(1.0, END)
            text1.insert(END, f"Error reading text file: {str(e)}")


def reveal_audio_message(audio_path):
    """Reveal a hidden message from an audio file."""
    audio = wave.open(audio_path, 'rb')
    frames = np.frombuffer(audio.readframes(audio.getnframes()), dtype=np.int16)
    audio.close()

    # Extract the LSBs from the audio frames
    binary_message = ''.join(str(frame & 1) for frame in frames)

    # Convert binary data to text
    chars = [binary_message[i:i + 8] for i in range(0, len(binary_message), 8)]
    decoded_message = ''.join([chr(int(char, 2)) for char in chars])

    # Extract message up to the delimiter
    return decoded_message.split("<<<EOF>>>")[0]

# Function to embed data into JPEG using DCT coefficients
def hide_data_in_jpeg(jpeg_path, message, output_path):
    # Convert the message to binary
    message_binary = ''.join(format(ord(c), '08b') for c in message) + "11111111"  # EOF marker

    # Read the JPEG file and access its DCT coefficients
    image = cv2.imread(jpeg_path, cv2.IMREAD_UNCHANGED)
    if image is None:
        raise ValueError("Invalid JPEG file.")

    # Convert to YCrCb color space and extract Y channel
    ycrcb = cv2.cvtColor(image, cv2.COLOR_BGR2YCrCb)
    y_channel = ycrcb[:, :, 0]

    # Apply DCT block-wise (8x8 blocks)
    height, width = y_channel.shape
    message_index = 0
    for i in range(0, height, 8):
        for j in range(0, width, 8):
            block = y_channel[i:i + 8, j:j + 8]
            dct_block = cv2.dct(np.float32(block) - 128)  # Apply DCT

            # Embed message bits in non-zero coefficients
            flat_block = dct_block.flatten()
            for k in range(len(flat_block)):
                if flat_block[k] != 0 and message_index < len(message_binary):
                    # Use round() to make it an integer, apply bitwise operation, then revert to float
                    coeff = int(round(flat_block[k]))
                    coeff = (coeff & ~1) | int(message_binary[message_index])  # Embed bit
                    flat_block[k] = float(coeff)  # Convert back to float
                    message_index += 1

            # Reconstruct the block
            dct_block = flat_block.reshape((8, 8))
            y_channel[i:i + 8, j:j + 8] = cv2.idct(dct_block) + 128  # Apply inverse DCT

            if message_index >= len(message_binary):
                break
        if message_index >= len(message_binary):
            break

    # Merge modified Y channel back and save the image
    ycrcb[:, :, 0] = y_channel
    modified_image = cv2.cvtColor(ycrcb, cv2.COLOR_YCrCb2BGR)
    cv2.imwrite(output_path, modified_image)


# Function to extract hidden data from JPEG
def reveal_data_in_jpeg(jpeg_path):
    # Read the JPEG file and access its DCT coefficients
    image = cv2.imread(jpeg_path, cv2.IMREAD_UNCHANGED)
    if image is None:
        raise ValueError("Invalid JPEG file.")

    ycrcb = cv2.cvtColor(image, cv2.COLOR_BGR2YCrCb)
    y_channel = ycrcb[:, :, 0]

    # Extract LSBs from DCT coefficients
    binary_message = ""
    for i in range(0, y_channel.shape[0], 8):
        for j in range(0, y_channel.shape[1], 8):
            block = y_channel[i:i + 8, j:j + 8]
            dct_block = cv2.dct(np.float32(block) - 128)

            for coeff in dct_block.flatten():
                if coeff != 0:
                    coeff = int(round(coeff))  # Ensure integer for bitwise operations
                    binary_message += str(coeff & 1)

    # Convert binary to string
    chars = [binary_message[i:i + 8] for i in range(0, len(binary_message), 8)]
    message = ''.join([chr(int(char, 2)) for char in chars])

    # Stop at EOF marker
    return message.split(chr(255))[0]

        def hide_audio_message(audio_path, message, output_path):
            """Hides a message in an audio file by modifying the LSB of audio samples."""
            message_binary = ''.join(format(ord(char), '08b') for char in message)
            message_length = len(message_binary)

            with wave.open(audio_path, 'rb') as audio:
                params = audio.getparams()
                num_channels = params.nchannels
                sample_width = params.sampwidth
                num_frames = params.nframes

                if sample_width != 2:
                    raise ValueError("Only 16-bit WAV files are supported.")

                frames = audio.readframes(num_frames)
                frame_list = list(struct.unpack("<" + "h" * num_frames * num_channels, frames))

            if len(message_binary) > len(frame_list):
                raise ValueError("Message is too long to fit in the selected audio file.")

            length_binary = format(message_length, '032b')
            for i in range(32):
                frame_list[i] = (frame_list[i] & ~1) | int(length_binary[i])

            for i in range(message_length):
                frame_list[32 + i] = (frame_list[32 + i] & ~1) | int(message_binary[i])

            modified_frames = struct.pack("<" + "h" * len(frame_list), *frame_list)

            with wave.open(output_path, 'wb') as modified_audio:
                modified_audio.setparams(params)
                modified_audio.writeframes(modified_frames)

        #
        def reveal_audio_message(audio_path):
            """Reveal a hidden message from an audio file."""
            audio = wave.open(audio_path, 'rb')
            frames = np.frombuffer(audio.readframes(audio.getnframes()), dtype=np.int16)
            audio.close()

            # Extract the LSBs from the audio frames
            binary_message = ''.join(str(frame & 1) for frame in frames)

            # Convert binary data to text
            chars = [binary_message[i:i + 8] for i in range(0, len(binary_message), 8)]
            decoded_message = ''.join([chr(int(char, 2)) for char in chars])

            # Extract message up to the delimiter
            return decoded_message.split("<<<EOF>>>")[0]
